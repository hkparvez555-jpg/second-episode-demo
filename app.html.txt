function sortedEps(){return [...state.episodes].sort((x,y)=>x.date<y.date?-1:1)}
function windowCounts(atISO){
  const eps=state.episodes.filter(e=>e.date<=atISO);
  return {six:eps.filter(e=>daysBetween(e.date,atISO)<=183).length, year:eps.filter(e=>daysBetween(e.date,atISO)<=365).length};
}
function isRecurrentAt(t){const c=windowCounts(t);return c.six>=2||c.year>=3}

function score(a){
  const s={OEST:0,CYCLE:0,EXPO:0,MIMIC:0};
  if(a.d1==='stopped'||a.d1==='hyst') s.OEST+=3;
  if(a.d1==='irregular'){s.OEST+=1;s.CYCLE+=1}
  if(a.d1==='regular') s.CYCLE+=2;
  if(a.d2==='50') s.OEST+=2; if(a.d2==='45') s.OEST+=1;
  if(a.d3==='often'){s.OEST+=3;s.MIMIC+=1} if(a.d3==='sometimes') s.OEST+=1;
  if(a.d4==='usually') s.CYCLE+=3; if(a.d4==='sometimes') s.CYCLE+=1;
  if(a.d5==='around'||a.d5==='after') s.CYCLE+=2;
  if(a.d6==='yes') s.CYCLE+=2;
  if(a.d7==='3-5') s.EXPO+=2; if(a.d7==='6+') s.EXPO+=3;
  if(a.d8==='yes') s.MIMIC+=3; if(a.d8==='never') s.EXPO+=2;
  if(a.d9==='often') s.MIMIC+=3; if(a.d9==='sometimes') s.MIMIC+=1;
  if(a.d10==='yes') s.EXPO+=3;
  // test results she already has
  for(const t of state.tests){ if(t.finding==='vaginal') s.MIMIC+=2; if(t.finding==='ph') s.OEST+=1; }
  return s;
}
function assign(a){
  const s=score(a), order=['MIMIC','OEST','EXPO','CYCLE'];
  const ranked=order.filter(k=>s[k]>=3).sort((x,y)=>s[y]-s[x]||order.indexOf(x)-order.indexOf(y));
  const primary=ranked[0]||'GENERAL', secondary=ranked[1]||null;
  const expoAddon=s.EXPO>=2&&primary!=='EXPO'&&secondary!=='EXPO';
  return {s,primary,secondary,expoAddon};
}
function buildPlan(){
  if(!state.assessment) return null;
  const a=state.assessment.answers, r=assign(a), bank=askBank(a);
  const main=asksFor(r.primary,a);
  const extra=[];
  // GIRFT urology A&G template 14: these apply to every woman with recurrent UTI, whatever her pattern
  const periPost=['irregular','stopped','hyst'].includes(a.d1)||a.d2==='45'||a.d2==='50';
  const core=['ultrasound', main.includes('cultureShare')?null:'culture', 'methenamine', ...(periPost?['vagOest','otc']:[])].filter(Boolean);
  for(const id of core) if(!main.includes(id)&&!extra.includes(id)) extra.push(id);
  if(r.secondary) for(const id of asksFor(r.secondary,a)) if(!main.includes(id)&&!extra.includes(id)) extra.push(id);
  if(r.expoAddon) for(const id of ['culture','methenamine']) if(!main.includes(id)&&!extra.includes(id)&&!(id==='culture'&&main.includes('cultureShare'))) extra.push(id);
  for(const id of state.extraAsks) if(!main.includes(id)&&!extra.includes(id)) extra.push(id);
  if(main.includes('cultureShare')){const i=extra.indexOf('culture'); if(i>=0) extra.splice(i,1);} // one culture ask only
  return {...r,main:main.map(id=>({id,...bank[id]})),extra:extra.slice(0,7).map(id=>({id,...bank[id]}))};
}

/* Bladder health score: a summary of her own logged record, not a prediction. Weights are provisional. */
function scoreAt(t){
  const eps=state.episodes.filter(e=>e.date<=t&&daysBetween(e.date,t)<=365);
  const rec=isRecurrentAt(t);
  const burden=Math.max(0,25-7*eps.length);
  const exposure=Math.max(0,25-6*eps.filter(e=>e.abx==='yes').length);
  let prevention=25, completeness;
  const asmtDone=state.assessment&&state.assessment.date<=t;
  const fus=Object.values(state.followups).filter(f=>f.date<=t);
  if(rec){
    prevention=0;
    const last=fus.sort((x,y)=>x.date<y.date?1:-1)[0];
    if(last&&last.asks){const v=Object.values(last.asks);prevention=Math.round(25*v.filter(x=>x==='started').length/Math.max(1,v.length))}
  }
  const sent=eps.length?eps.filter(e=>e.sample==='yes').length/eps.length:1;
  completeness=Math.round((rec?(asmtDone?8:0):8)+9*sent+(rec?(fus.length?8:0):8));
  const parts={burden,exposure,prevention,completeness};
  return {total:burden+exposure+prevention+completeness,parts};
}
function scoreHistory(){
  const pts=new Set(state.episodes.map(e=>e.date));
  if(state.assessment) pts.add(state.assessment.date);
  Object.values(state.followups).forEach(f=>pts.add(f.date));
  const t=iso(today()); pts.add(t);
  return [...pts].filter(d=>daysBetween(d,t)<=400).sort().map(d=>({d,s:scoreAt(d).total}));
}
function frozenScore(){
  if(!state.redFlag) return null;
  const before=addDays(state.redFlag.date,-1);
  return scoreAt(before).total;
}

   case 'stiStart': sti={step:'safety',i:0,a:{},flags:new Set()}; render(true); break;
   case 'stiFlag': sti.flags.has(el.dataset.id)?sti.flags.delete(el.dataset.id):sti.flags.add(el.dataset.id); render(); break;
   case 'stiSafetyDone': sti.step='q'; sti.i=0; render(true); break;
   case 'stiMulti': {const s=sti.a.sym||(sti.a.sym=[]);const v=el.dataset.v;const k=s.indexOf(v);k>=0?s.splice(k,1):s.push(v);render();break}
   case 'stiAnswer':
     sti.a[el.dataset.q]=el.dataset.v;
     el.closest('.opts').querySelectorAll('.opt').forEach(o=>o.classList.toggle('on',o===el));
     setTimeout(stiAdvance,260); break;
   case 'stiNext': if(!sti.a.sym) sti.a.sym=[]; stiAdvance(); break;
   case 'stiBack': if(sti.i>0){sti.i--}else{sti.step='safety'} render(true); break;
   case 'stiReset': sti=null; render(true); break;
  }
});
function stiAdvance(){
  const qs=stiVisible();
  if(sti.i<qs.length-1){sti.i++}else{sti.step='result'}
  render(true);
}
